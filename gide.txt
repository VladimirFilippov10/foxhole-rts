    // ===== ПЕХОТНЫЙ МЕДИК =====
    if (u.isMedic && u.hp > 0) {
        const now = this.time;
        if (now - (u._lastHealTick || 0) >= (u.healTickInterval || 30)) {
            u._lastHealTick = now;
            const range = u.healRange || 80;
            let target = null, bestDist = Infinity;
            for (const o of this.units) {
                if (o === u || o.faction !== u.faction || o.hp <= 0) continue;
                if (!isInfUnit(o)) continue;
                if (o.hp >= o.maxHp - 0.5) continue;
                const d = dist(u.x, u.y, o.x, o.y);
                if (d <= range && d < bestDist) { bestDist = d; target = o; }
            }
            if (target && typeof u.healUnit === 'function') {
                u.healUnit(target);
            }
        }
    }

И в класс медика (B3) добавь метод:

    healUnit(targetUnit) {
        if (!targetUnit || typeof targetUnit !== 'object') return 0;
        if (targetUnit.isInfantry !== true
            && targetUnit.type !== 'infantry') return 0;
        const hp = Number(targetUnit.hp);
        const maxHp = Number(targetUnit.maxHp);
        if (!Number.isFinite(hp) || !Number.isFinite(maxHp)) return 0;
        if (hp >= maxHp - 0.5) return 0;
        const amount = Math.min(this.healRate || 0.08, maxHp - hp);
        if (!Number.isFinite(amount) || amount <= 0) return 0;
        targetUnit.hp = hp + amount;
        return amount;
    }

    pickInfantryType(role) {
        const isWardens = this.G.ai === 'wardens';
        const medicType = isWardens ? 'medic_varden' : 'medic_colonist';
        const r = Math.random();
        if (role === 'defense') {
            return r < 0.70 ? 'mg_gunner'
                 : (r < 0.90 ? 'rifleman' : medicType);
        }
        if (r < 0.35) return 'rifleman';
        if (r < 0.55) return 'grenadier';
        if (r < 0.70) return 'mg_gunner';
        if (r < 0.80) return 'assualt_varden';
        if (r < 0.88) return 'sniper';
        if (r < 0.95) return 'my_infantry';
        return medicType;
    }

================================================================================
  ЧАСТЫЕ ОШИБКИ
================================================================================

  Симптом                                    Причина
  ------------------------------------------------------------------------
  Карточки нет в меню завода                 не добавлен в FACTION_VEHICLES
                                             (техника) / FACTION_INFANTRY (пехота)

  "Недостаточно ресурсов" всегда             нет VEHICLE_COSTS[type]

  Юнит = базовый (HP 100, dmg 10)            нет map[type] (техника)
                                             / case (пехота)

  Не рисуется (прямоугольник)                нет ветки в renderer.js -> drawUnit

  TypeError: Cannot read properties          VEHICLE_COSTS[type] = undefined
  of undefined

  Пехота не производится                     не добавлен в INFANTRY_TYPES
  (упирается в лимит техники)                (vehicles.js)

  ИИ не строит                               не добавлен в vehicleRoster
                                             / pickInfantryType

  Медик не лечит                             нет isMedic / healUnit()
                                             ИЛИ нет блока B12 в game.js

  Артиллерия не стреляет через ИИ            нет isArtillery = true
                                             / isMortar = true

  Артиллерия в лоб едет на базу              нет minRange или isArtillery
